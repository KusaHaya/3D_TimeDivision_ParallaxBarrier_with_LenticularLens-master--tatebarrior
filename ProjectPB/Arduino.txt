#define REFRESH_RATE 144
#define HIGH_RATE 100

#define CONNECTION_BEGIN 10
#define CONNECTION_END 11
#define NEXT_FRAME 1
#define ENABLE_TIMEDIVISION 12
#define DISABLE_TIMEDIVISION 13          

// 8出力用 (個別に光らせたい場合：100~107)
#define LIGHT_CON_0 100
#define LIGHT_CON_1 101
#define LIGHT_CON_2 102
#define LIGHT_CON_3 103
#define LIGHT_CON_4 104
#define LIGHT_CON_5 105
#define LIGHT_CON_6 106
#define LIGHT_CON_7 107
#define LIGHT_EXIT 110

#define NEXT_TIMEDIVISION 21
#define RESET_SYNC 20

// 4時分割 同時制御用コマンド (新規追加)
#define TIME_DIV_0 200
#define TIME_DIV_1 201
#define TIME_DIV_2 202
#define TIME_DIV_3 203

// PC画面のkkと対応するバックライト位相を連続点灯で固定するコマンド
#define HOLD_FRAME_0 210
#define HOLD_FRAME_1 211
#define HOLD_FRAME_2 212
#define HOLD_FRAME_3 213
int PIN_PAST = -1;
int cnt = 0;
const int offset = 1500;

// 8出力用に配列サイズとピン番号を変更
// !!! ピン番号はご自身の環境に合わせてください !!!
// (例： 9-12が上部、2-5が下部)
const int OUTPUT_PINS[8] = {9, 10, 11, 12, 4, 5, 6, 7};

// 144 Hzの1フレームは約6944us。V-Sync起点で上下を時間差点灯する。
const unsigned long TOP_ON_US = 2080;
const unsigned long BOTTOM_SWITCH_US = 3750;
const unsigned long BACKLIGHT_OFF_US = 6250;

enum BacklightStage {
  BACKLIGHT_IDLE,
  WAIT_TOP_ON,
  WAIT_BOTTOM_SWITCH,
  WAIT_BACKLIGHT_OFF
};

unsigned long backlightFrameStartedAt = 0;
BacklightStage backlightStage = BACKLIGHT_IDLE;
int timedPastBottomPin = -1;
int timedTopPin = -1;
int timedBottomPin = -1;

const float pluse_width = 1000.0 / REFRESH_RATE - offset / 1000;
const float pluse_width_tb = 1000.0 / REFRESH_RATE ;
const float delay_time = (100.0 - HIGH_RATE) * pluse_width / 100 / 2;

void setup()
{
  Serial.begin(115200);
  for (int i = 0; i < 8; i++)
  {
    pinMode(OUTPUT_PINS[i], OUTPUT);
  }
  delay(10);
  for (int i = 0; i < 8; i++)
  {
    digitalWrite(OUTPUT_PINS[i], HIGH);
  }
}

void turnAllLightsOff()
{
  for (int i = 0; i < 8; i++) {
    digitalWrite(OUTPUT_PINS[i], LOW);
  }
}

// 既存：1ピン制御用
void onDelay(int PIN)
{
  delayMicroseconds(offset);
  digitalWrite(PIN, HIGH);
  delayMicroseconds(pluse_width * HIGH_RATE / 100 * 1000);
  digitalWrite(PIN, LOW);
}

// 新規追加：上下ペア同時制御用
void onDelaySimultaneous(int PIN_TOP, int PIN_BOTTOM)
{
  if (PIN_PAST >= 0) {
    digitalWrite(PIN_PAST, HIGH);
  }

  delayMicroseconds(2500);

  digitalWrite(PIN_TOP, HIGH);
  delayMicroseconds(2000);

  if (PIN_PAST >= 0) {
    digitalWrite(PIN_PAST, LOW);
  }

  digitalWrite(PIN_BOTTOM, HIGH);

  delayMicroseconds(pluse_width_tb * 1000 - 4800);

  digitalWrite(PIN_TOP, LOW);

  PIN_PAST = PIN_BOTTOM;
  digitalWrite(PIN_BOTTOM, LOW);

  delayMicroseconds(200);
}

void runOneTimeDivisionCycle()
{
  // 旧PC送信順と同じ: 0 → 3 → 2 → 1
  const int phaseOrder[4] = {0, 3, 2, 1};

  for (int i = 0; i < 4; i++) {
    int phase = phaseOrder[i];
    cnt = phase;

    onDelaySimultaneous(

      OUTPUT_PINS[phase + 4],
      OUTPUT_PINS[phase]
    );
  }
}

void runOneTimeDivisionFrame(int frameIndex)
{
  // PCの表示順kk=0,1,2,3を、実際の点灯順0,3,2,1へ変換する。
  const int phaseOrder[4] = {0, 3, 2, 1};
  int phase = phaseOrder[frameIndex & 3];

  timedPastBottomPin = PIN_PAST;
  timedTopPin = OUTPUT_PINS[phase + 4];
  timedBottomPin = OUTPUT_PINS[phase];

  turnAllLightsOff();
  if (timedPastBottomPin >= 0) {
    digitalWrite(timedPastBottomPin, HIGH);
  }

  cnt = phase;
  PIN_PAST = timedBottomPin;
  backlightFrameStartedAt = micros();
  backlightStage = WAIT_TOP_ON;
}

int prev = 0;
bool running = true;

void loop()
{
  if (Serial.available() > 0)
  {
    int incomingByte = Serial.read();

    switch (incomingByte)
    {
    case CONNECTION_BEGIN:
      for (int i = 0; i < 8; i++) {
        if (cnt != i) { digitalWrite(OUTPUT_PINS[i], LOW); }
      }
      break;
    case LIGHT_EXIT:
      backlightStage = BACKLIGHT_IDLE;
      PIN_PAST = -1;
      for (int i = 0; i < 8; i++) {
        digitalWrite(OUTPUT_PINS[i], HIGH);
      }
      break;

    // (オプション) 8ピン個別テスト用
    case LIGHT_CON_0:
    case LIGHT_CON_1:
    case LIGHT_CON_2:
    case LIGHT_CON_3:
    case LIGHT_CON_4:
    case LIGHT_CON_5:
    case LIGHT_CON_6:
    case LIGHT_CON_7:
    //  int received_kk = incomingByte - LIGHT_CON_0;
    //  if (received_kk >= 0 && received_kk < 8) {
    //    turnAllLightsOff();
    //    onDelay(OUTPUT_PINS[received_kk]);
   //   }
      break;

    // PCから各表示フレームごとに1コマンドを受け、V-Sync起点で
    // 前フレーム下側→現フレーム上側→現フレーム下側の順に点灯する。
  case TIME_DIV_0:
  case TIME_DIV_1:
  case TIME_DIV_2:
  case TIME_DIV_3:
  {
    if (running) {
      int frameIndex = incomingByte - TIME_DIV_0;
      runOneTimeDivisionFrame(frameIndex);
    }
    break;
  }

  case HOLD_FRAME_0:
  case HOLD_FRAME_1:
  case HOLD_FRAME_2:
  case HOLD_FRAME_3:
  {
    // PCの表示順kk=0,1,2,3を、実際の点灯順0,3,2,1へ変換する。
    // 時分割を停止し、選択された上下ペアを連続点灯する。
    const int phaseOrder[4] = {0, 3, 2, 1};
    int frameIndex = incomingByte - HOLD_FRAME_0;
    int phase = phaseOrder[frameIndex];

    running = false;
    backlightStage = BACKLIGHT_IDLE;
    cnt = phase;
    PIN_PAST = -1;
    turnAllLightsOff();
    digitalWrite(OUTPUT_PINS[phase + 4], HIGH);
    digitalWrite(OUTPUT_PINS[phase], HIGH);
    break;
  }

    case NEXT_TIMEDIVISION:
    {
    if (running) {
        backlightStage = BACKLIGHT_IDLE;
        cnt = (cnt + 3) & 3;  // 以前の相順 0→3→2→1 を維持
        turnAllLightsOff();
        onDelaySimultaneous(OUTPUT_PINS[cnt + 4], OUTPUT_PINS[cnt]);
    }
    break;
    }

  
    // --- 制御ここまで ---

    case RESET_SYNC:
      running = false;
      backlightStage = BACKLIGHT_IDLE;
      cnt = 0;
      PIN_PAST = -1;
      timedPastBottomPin = -1;
      timedTopPin = -1;
      timedBottomPin = -1;
      turnAllLightsOff();
      break;

    case ENABLE_TIMEDIVISION:
      running = true;
      break;
    case DISABLE_TIMEDIVISION:
      running = false;
      backlightStage = BACKLIGHT_IDLE;
      digitalWrite(OUTPUT_PINS[cnt], HIGH);
      break;
    }
  }

  // Non-blocking scan timing. Multiple stages may be applied in one loop if
  // processing was briefly delayed; the next serial command remains responsive.
  unsigned long elapsedUs = micros() - backlightFrameStartedAt;

  if (backlightStage == WAIT_TOP_ON && elapsedUs >= TOP_ON_US) {
    digitalWrite(timedTopPin, HIGH);
    backlightStage = WAIT_BOTTOM_SWITCH;
  }

  if (backlightStage == WAIT_BOTTOM_SWITCH && elapsedUs >= BOTTOM_SWITCH_US) {
    if (timedPastBottomPin >= 0) {
      digitalWrite(timedPastBottomPin, LOW);
    }
    digitalWrite(timedBottomPin, HIGH);
    backlightStage = WAIT_BACKLIGHT_OFF;
  }

  if (backlightStage == WAIT_BACKLIGHT_OFF && elapsedUs >= BACKLIGHT_OFF_US) {
    digitalWrite(timedTopPin, LOW);
    digitalWrite(timedBottomPin, LOW);
    backlightStage = BACKLIGHT_IDLE;
  }
}
