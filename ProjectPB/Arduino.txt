// Non-blocking backlight timing from the 2026-09-29 progress report.
// Must match the actual Windows display refresh rate: 144 or 120 Hz.
#ifndef REFRESH_RATE
#define REFRESH_RATE 144
#endif
#define CONNECTION_BEGIN 10
#define CONNECTION_END 11
#define ENABLE_TIMEDIVISION 12
#define DISABLE_TIMEDIVISION 13
#define RESET_SYNC 20
#define NEXT_TIMEDIVISION 21
#define LIGHT_CON_0 100
#define LIGHT_CON_7 107
#define LIGHT_EXIT 110
#define TIME_DIV_0 200
#define TIME_DIV_3 203
#define HOLD_FRAME_0 210
#define HOLD_FRAME_3 213

const int OUTPUT_PINS[8] = {9, 10, 11, 12, 4, 5, 6, 7};
// Existing wiring convention: phase+4 = TOP, phase = BOTTOM.
#if REFRESH_RATE == 144
const unsigned long TOP_ON_US = 2083;
const unsigned long BOTTOM_SWITCH_US = 3750;
const unsigned long ALL_OFF_US = 6250;
#elif REFRESH_RATE == 120
const unsigned long TOP_ON_US = 2500;
const unsigned long BOTTOM_SWITCH_US = 4500;
const unsigned long ALL_OFF_US = 7500;
#else
#error Set REFRESH_RATE to 120 or 144 to use a validated timing table.
#endif

bool running = true;
bool pulseActive = false;
bool topStarted = false;
bool bottomSwitched = false;
int cnt = 0;
int previousPhase = -1;
unsigned long frameStartUs = 0;

void turnAllLightsOff() {
  for (int i=0; i<8; ++i) digitalWrite(OUTPUT_PINS[i], LOW);
}
void cancelPulse() {
  pulseActive = false;
  topStarted = false;
  bottomSwitched = false;
}
void resetBacklight() {
  cancelPulse();
  turnAllLightsOff();
  previousPhase = -1;
  cnt = 0;
}
void holdPhase(int phase) {
  cancelPulse();
  running = false;
  turnAllLightsOff();
  cnt = phase & 3;
  previousPhase = -1;
  digitalWrite(OUTPUT_PINS[cnt+4], HIGH);
  digitalWrite(OUTPUT_PINS[cnt], HIGH);
}
void startPhase(int phase) {
  turnAllLightsOff();
  cnt = phase & 3;
  frameStartUs = micros();
  topStarted = false;
  bottomSwitched = false;
  pulseActive = true;
  // Re-light the preceding phase's bottom at the next frame command.
  if (previousPhase >= 0) digitalWrite(OUTPUT_PINS[previousPhase], HIGH);
}
void updateBacklight() {
  if (!pulseActive) return;
  // Unsigned subtraction also handles micros() rollover.
  const unsigned long elapsed = micros() - frameStartUs;
  if (elapsed >= ALL_OFF_US) {
    turnAllLightsOff();
    pulseActive = false;
    previousPhase = cnt;
    return;
  }
  if (!topStarted && elapsed >= TOP_ON_US) {
    digitalWrite(OUTPUT_PINS[cnt+4], HIGH);
    topStarted = true;
  }
  if (!bottomSwitched && elapsed >= BOTTOM_SWITCH_US) {
    if (previousPhase >= 0) digitalWrite(OUTPUT_PINS[previousPhase], LOW);
    digitalWrite(OUTPUT_PINS[cnt], HIGH);
    bottomSwitched = true;
  }
}
void handleCommand(int command) {
  if (command >= TIME_DIV_0 && command <= TIME_DIV_3) {
    if (running) {
      // If a command arrives early, the last commanded phase becomes PREV.
      if (pulseActive) previousPhase = cnt;
      startPhase(command - TIME_DIV_0);
    }
    return;
  }
  if (command >= HOLD_FRAME_0 && command <= HOLD_FRAME_3) {
    holdPhase(command - HOLD_FRAME_0);
    return;
  }
  if (command >= LIGHT_CON_0 && command <= LIGHT_CON_7) {
    cancelPulse(); running = false; turnAllLightsOff();
    previousPhase = -1;
    digitalWrite(OUTPUT_PINS[command-LIGHT_CON_0], HIGH);
    return;
  }
  switch(command) {
    case RESET_SYNC: resetBacklight(); break;
    case ENABLE_TIMEDIVISION: resetBacklight(); running = true; break;
    case DISABLE_TIMEDIVISION: holdPhase(cnt); break;
    case NEXT_TIMEDIVISION:
      if (running) {
        if (pulseActive) previousPhase = cnt;
        startPhase((cnt+3)&3);
      }
      break;
    case CONNECTION_BEGIN:
    case CONNECTION_END:
      resetBacklight(); running = false; break;
    case LIGHT_EXIT:
      cancelPulse(); running = false; previousPhase = -1;
      for(int i=0;i<8;++i) digitalWrite(OUTPUT_PINS[i],HIGH);
      break;
  }
}
void setup() {
  Serial.begin(115200);
  for(int i=0;i<8;++i) pinMode(OUTPUT_PINS[i],OUTPUT);
  resetBacklight();
}
void loop() {
  updateBacklight();
  if (Serial.available()>0) handleCommand(Serial.read());
  updateBacklight();
}
