cmake_minimum_required(VERSION 3.24)
project(ProjectPBDisplayXR VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_library(projectpb_core STATIC src/Tracking.cpp)
target_include_directories(projectpb_core PUBLIC include)
if(MSVC)
  target_compile_options(projectpb_core PRIVATE /W4 /permissive- /utf-8)
endif()
if(BUILD_TESTING)
  add_executable(projectpb_core_tests tests/core_tests.cpp)
  target_link_libraries(projectpb_core_tests PRIVATE projectpb_core)
  add_test(NAME projectpb_core COMMAND projectpb_core_tests)
endif()
option(PROJECTPB_BUILD_PLUGIN "Build the Windows D3D11 hold-phase plugin" ON)
set(DISPLAYXR_RUNTIME_SOURCE_DIR "" CACHE PATH "Official runtime checkout (headers only; not modified)")
set(PROJECTPB_RUNTIME_SHA ac3ae19215bbcfc6aa091224d109a5b113c1c8dd)
if(PROJECTPB_BUILD_PLUGIN)
  if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The plugin currently requires Windows x64")
  endif()
  find_package(Git REQUIRED)
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${DISPLAYXR_RUNTIME_SOURCE_DIR}" rev-parse HEAD
    OUTPUT_VARIABLE runtime_sha OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE git_result)
  if(NOT git_result EQUAL 0 OR NOT runtime_sha STREQUAL PROJECTPB_RUNTIME_SHA)
    message(FATAL_ERROR "Use an official runtime checkout pinned to ${PROJECTPB_RUNTIME_SHA}")
  endif()
  set(runtime_includes "${DISPLAYXR_RUNTIME_SOURCE_DIR}/src/xrt/include")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/interleave.hlsl")
  file(READ shaders/interleave.hlsl PROJECTPB_SHADER)
  configure_file(src/ShaderSource.h.in generated/ShaderSource.h @ONLY)
  add_library(projectpb_abi INTERFACE)
  target_include_directories(projectpb_abi INTERFACE "${runtime_includes}")
  add_library(DisplayXR-ProjectPB SHARED src/Plugin.cpp src/Device.cpp src/Processor.cpp src/OscReceiver.cpp)
  target_link_libraries(DisplayXR-ProjectPB PRIVATE projectpb_core projectpb_abi d3d11 d3dcompiler dxgi ws2_32)
  target_include_directories(DisplayXR-ProjectPB PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" src)
  target_compile_definitions(DisplayXR-ProjectPB PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  if(MSVC)
    target_compile_options(DisplayXR-ProjectPB PRIVATE /W4 /permissive- /utf-8)
  endif()
  if(BUILD_TESTING)
    add_executable(projectpb_abi_gpu_tests tests/abi_gpu_tests.cpp src/OscReceiver.cpp)
    target_include_directories(projectpb_abi_gpu_tests PRIVATE src)
    target_compile_definitions(projectpb_abi_gpu_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(projectpb_abi_gpu_tests PRIVATE projectpb_core projectpb_abi d3d11 dxgi ws2_32)
    if(MSVC)
      target_compile_options(projectpb_abi_gpu_tests PRIVATE /W4 /permissive- /utf-8)
      target_compile_options(projectpb_core_tests PRIVATE /W4 /permissive- /utf-8)
    endif()
    add_test(NAME projectpb_abi_gpu COMMAND projectpb_abi_gpu_tests "$<TARGET_FILE:DisplayXR-ProjectPB>")
  endif()
endif()
